\subsubsection{Normal (Gaussian)}
\label{subsubsec:prob-stat:dist:norm}

\paragraph{PDF}
\label{para:prob-stat:dist:norm:pdf}
\begin{equation}
    \label{eq:norm-pdf}
    p(x;\, \mu, \sigma^2) = \frac{1}{\sigma\sqrt{2\pi}}\exp\!\left(-\frac{(x-\mu)^2}{2\sigma^2}\right)
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:norm:cdf}
\begin{equation}
    \label{eq:norm-cdf}
    \Phi(x;\, \mu, \sigma^2) = P(X \leq x) = \frac{1}{2}\left[1 + \operatorname{erf}\!\left(\frac{x - \mu}{\sigma\sqrt{2}}\right)\right]
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:norm:mean}
\begin{equation}
    \label{eq:norm-mean}
    \mathbb{E}[X] = \mu
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:norm:variance}
\begin{equation}
    \label{eq:norm-variance}
    \operatorname{Var}(X) = \sigma^2
\end{equation}

\subsubsection{Exponential}
\label{subsubsec:prob-stat:dist:exp}

\paragraph{PDF}
\label{para:prob-stat:dist:exp:pdf}
\begin{equation}
    \label{eq:exp-pdf}
    p(x; \lambda) =
    \begin{cases}
        \lambda e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:exp:cdf}
\begin{equation}
    \label{eq:exp-cdf}
    P(x; \lambda) = P(X \leq x) =
    \begin{cases}
        1 - e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:exp:mean}
\begin{equation}
    \label{eq:exp-mean}
    \mathbb{E}[X] = \frac{1}{\lambda}
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:exp:variance}
\begin{equation}
    \label{eq:exp-variance}
    \operatorname{Var}(X) = \frac{1}{\lambda^2}
\end{equation}

\subsubsection{Poisson}
\label{subsubsec:prob-stat:dist:poiss}

\paragraph{PMF}
\label{para:prob-stat:dist:poiss:pmf}
\begin{equation}
    \label{eq:poiss-pmf}
    f(k; \lambda) = P(X=k) = \frac{\lambda^k e^{-\lambda}}{k!}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:poiss:cdf}
\begin{equation}
    \label{eq:poiss-cdf}
    e^{-\lambda} \sum_{j=0}^{\lfloor k \rfloor} \frac{\lambda^j}{j!}
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:poiss:mean}
\begin{equation}
    \label{eq:poiss-mean}
    \mathbb{E}[X] = \lambda
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:poiss:variance}
\begin{equation}
    \label{eq:poiss-variance}
    \operatorname{Var}(X) = \lambda
\end{equation}

\subsubsection{Bernoulli}
\label{subsubsec:prob-stat:dist:bern}

\paragraph{PMF}
\label{para:prob-stat:dist:bern:pmf}
\begin{equation}
    \label{eq:bern-pmf}
    P(X = x) =
    \begin{cases}
        1 - p, & x = 0 \\
        p, & x = 1
    \end{cases}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:bern:cdf}
\begin{equation}
    \label{eq:bern-cdf}
    P(X \le x) =
    \begin{cases}
        0, & x < 0 \\
        1-p, & 0 \le x < 1 \\
        1, & x \ge 1
    \end{cases}
\end{equation}

\paragraph{Mean}\label{para:prob-stat:dist:bern:mean}
\begin{equation}
    \label{eq:bern-mean}
    \mathbb{E}[X] = 0 \times (1 - p) + 1 \times p = p
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:bern:variance}
\begin{equation}
    \label{eq:bern-variance}
    \operatorname{Var}(X) = p (1-p)
\end{equation}

\subsubsection{Softmax Probabilities}\label{subsubsec:prob-stat:softmax:prob}
\begin{equation}
    \label{eq:softmax}
    \sigma(\mathbf{z})_i = \frac{e^{z_i}}{\sum_{j=1}^{K} e^{z_j}}
\end{equation}

\subsubsection{Logarithm of Softmax Probabilities}\label{subsubsec:prob-stat:softmax:log}
\begin{equation}
    \label{eq:log-softmax}
    \log \sigma(\mathbf{z})_i = z_i - \log \sum_{j=1}^{K} e^{z_j}
\end{equation}

\subsubsection{Stable Softmax}\label{subsubsec:prob-stat:softmax:stable}
\begin{equation}
    \label{eq:stable-softmax}
    \sigma(\mathbf{z})_i = \frac{e^{z_i - \max_k z_k}}{\sum_{j=1}^{K} e^{z_j - \max_k z_k}}
\end{equation}

\section{Numpy}
\label{sec:numpy}

\begin{itemize}
    \item \texttt{np.array(lst)} turns a list into a numpy array; \texttt{arr.tolist()} turns a numpy array into a list.
    \item \texttt{np.random.choice(elements, p=probabilities)} draws samples from an array using specified probabilities.
    \item \texttt{np.random.seed(seed)} seeds the random number generator to ensure reproducible results.
    \item \texttt{np.random.permutation(n)} randomly permutes \texttt{n} indices; e.g., \texttt{n = X.shape[0]} for rows.
    \item Use \texttt{is None} and \texttt{is not None} to check whether a variable is \texttt{None}.
    \item If \texttt{a} and \texttt{b} are two NumPy arrays, \texttt{a + b} computes the element-wise sum.
    \item \texttt{np.sum(arr, axis=k)} sums \texttt{arr} along axis \texttt{k}; omit \texttt{axis} to sum all elements.
    \item \texttt{np.sqrt(arr)} applies element-wise square root; negative inputs yield \texttt{nan}.
    \item \texttt{arr.min()}, \texttt{arr.max()}, \texttt{arr.std()}, and \texttt{arr.mean()} reduce over all elements.
    \item Give those reductions \texttt{axis=0} for a column-wise result, or \texttt{axis=1} for a row-wise result.
    \item \texttt{np.reshape(a, shape)} returns \texttt{a} reshaped without altering it; \texttt{shape} is int or tuple.
    \item One entry of \texttt{shape} may be \texttt{-1}, and NumPy infers it from the array size.
    \item \texttt{np.cov(a)} treats each row of \texttt{a} as a variable, so an $m \times n$ input gives $m \times m$.
    \item \texttt{np.cov} normalizes by $n-1$; use \texttt{bias=True} for $n$, \texttt{rowvar=False} for columns.
\end{itemize}

\subsubsection{Naive Bayes}
\label{subsubsec:ml:gla:nb}
The Naive Bayes classifier relies on the naive assumption that features are independent of one another. Suppose there are $K$ classes to classify the variable $y$, given feature vector $\mathbf{x} = (x_1, \cdots, x_n)$. Based on Naive Bayes, the corresponding classifier is the function that assigns a class label $\hat{y} = C_k$ for some $k$, $1 \leq k \leq K$, as follows:
\begin{equation}
\label{eq:naive-bayes}
    \hat{y} = \operatorname*{argmax}_{1 \leq k \leq K} p(C_k) \prod_{i=1}^{n} p(x_i \mid C_k).
\end{equation}

\subsubsection{Laplace Smoothing}
\label{subsubsec:ml:gla:laplace-smoothing}
Laplace smoothing (additive smoothing) prevents zero-probability estimates when a feature value does not appear in the training data for a given class. For a discrete feature $x_i$ taking value $v$, the smoothed class-conditional probability is:
\begin{equation}
    \label{eq:laplace-class-cond}
    \hat{P}(x_i = v \mid C_k) = \frac{N_{k,i,v} + \alpha}{M_k + \alpha\, d_i},
\end{equation}
where $N_{k,i,v} = \#\{m : x_i^{(m)} = v,\, y^{(m)} = C_k\}$ is the number of training examples in class $C_k$ for which feature $x_i$ equals $v$, $M_k$ is the total number of training examples in class $C_k$, $d_i = \lvert V_i \rvert$ is the number of distinct values feature $x_i$ can take, and $\alpha > 0$ is the smoothing parameter ($\alpha = 1$ for Laplace smoothing; $0 < \alpha < 1$ for Lidstone smoothing).

The smoothed class prior is:
\begin{equation}
    \label{eq:laplace-prior}
    \hat{P}(C_k) = \frac{M_k + \alpha}{M + \alpha K},
\end{equation}
where $M$ is the total number of training examples and $K$ is the number of classes.
