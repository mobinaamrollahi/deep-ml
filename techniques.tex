\subsubsection{Normal (Gaussian)}
\label{subsubsec:prob-stat:dist:norm}

\paragraph{PDF}
\label{para:prob-stat:dist:norm:pdf}
\begin{equation}
    \label{eq:norm-pdf}
    p(x;\, \mu, \sigma^2) = \frac{1}{\sigma\sqrt{2\pi}}\exp\!\left(-\frac{(x-\mu)^2}{2\sigma^2}\right)
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:norm:cdf}
\begin{equation}
    \label{eq:norm-cdf}
    \Phi(x;\, \mu, \sigma^2) = P(X \leq x) = \frac{1}{2}\left[1 + \operatorname{erf}\!\left(\frac{x - \mu}{\sigma\sqrt{2}}\right)\right]
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:norm:mean}
\begin{equation}
    \label{eq:norm-mean}
    \mathbb{E}[X] = \mu
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:norm:variance}
\begin{equation}
    \label{eq:norm-variance}
    \operatorname{Var}(X) = \sigma^2
\end{equation}

\subsubsection{Exponential}
\label{subsubsec:prob-stat:dist:exp}

\paragraph{PDF}
\label{para:prob-stat:dist:exp:pdf}
\begin{equation}
    \label{eq:exp-pdf}
    p(x; \lambda) =
    \begin{cases}
        \lambda e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:exp:cdf}
\begin{equation}
    \label{eq:exp-cdf}
    P(x; \lambda) = P(X \leq x) =
    \begin{cases}
        1 - e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:exp:mean}
\begin{equation}
    \label{eq:exp-mean}
    \mathbb{E}[X] = \frac{1}{\lambda}
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:exp:variance}
\begin{equation}
    \label{eq:exp-variance}
    \operatorname{Var}(X) = \frac{1}{\lambda^2}
\end{equation}

\subsubsection{Poisson}
\label{subsubsec:prob-stat:dist:poiss}

\paragraph{PMF}
\label{para:prob-stat:dist:poiss:pmf}
\begin{equation}
    \label{eq:poiss-pmf}
    f(k; \lambda) = P(X=k) = \frac{\lambda^k e^{-\lambda}}{k!}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:poiss:cdf}
\begin{equation}
    \label{eq:poiss-cdf}
    e^{-\lambda} \sum_{j=0}^{\lfloor k \rfloor} \frac{\lambda^j}{j!}
\end{equation}

\paragraph{Mean}
\label{para:prob-stat:dist:poiss:mean}
\begin{equation}
    \label{eq:poiss-mean}
    \mathbb{E}[X] = \lambda
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:poiss:variance}
\begin{equation}
    \label{eq:poiss-variance}
    \operatorname{Var}(X) = \lambda
\end{equation}

\subsubsection{Bernoulli}
\label{subsubsec:prob-stat:dist:bern}

\paragraph{PMF}
\label{para:prob-stat:dist:bern:pmf}
\begin{equation}
    \label{eq:bern-pmf}
    P(X = x) =
    \begin{cases}
        1 - p, & x = 0 \\
        p, & x = 1
    \end{cases}
\end{equation}

\paragraph{CDF}
\label{para:prob-stat:dist:bern:cdf}
\begin{equation}
    \label{eq:bern-cdf}
    P(X \le x) =
    \begin{cases}
        0, & x < 0 \\
        1-p, & 0 \le x < 1 \\
        1, & x \ge 1
    \end{cases}
\end{equation}

\paragraph{Mean}\label{para:prob-stat:dist:bern:mean}
\begin{equation}
    \label{eq:bern-mean}
    \mathbb{E}[X] = 0 \times (1 - p) + 1 \times p = p
\end{equation}

\paragraph{Variance}
\label{para:prob-stat:dist:bern:variance}
\begin{equation}
    \label{eq:bern-variance}
    \operatorname{Var}(X) = p (1-p)
\end{equation}

\subsection{Softmax}\label{subsec:prob-stat:softmax}
\subsubsection{Softmax Probabilities}\label{subsubsec:prob-stat:softmax:prob}
\begin{equation}
    \label{eq:softmax}
    \sigma(\mathbf{z})_i = \frac{e^{z_i}}{\sum_{j=1}^{K} e^{z_j}}
\end{equation}

\subsubsection{Logarithm of Softmax Probabilities}\label{subsubsec:prob-stat:softmax:log}
\begin{equation}
    \label{eq:log-softmax}
    \log \sigma(\mathbf{z})_i = z_i - \log \sum_{j=1}^{K} e^{z_j}
\end{equation}

\subsubsection{Stable Softmax}\label{subsubsec:prob-stat:softmax:stable}
\begin{equation}
    \label{eq:stable-softmax}
    \sigma(\mathbf{z})_i = \frac{e^{z_i - \max_k z_k}}{\sum_{j=1}^{K} e^{z_j - \max_k z_k}}
\end{equation}

\subsubsection{Naive Bayes}
\label{subsubsec:ml:gla:nb}
The Naive Bayes classifier relies on the naive assumption that features are independent of one another. Suppose there are $K$ classes to classify the variable $y$, given feature vector $\mathbf{x} = (x_1, \cdots, x_n)$. Based on Naive Bayes, the corresponding classifier is the function that assigns a class label $\hat{y} = C_k$ for some $k$, $1 \leq k \leq K$, as follows:
\begin{equation}
\label{eq:naive-bayes}
    \hat{y} = \operatorname*{argmax}_{1 \leq k \leq K} p(C_k) \prod_{i=1}^{n} p(x_i \mid C_k).
\end{equation}
