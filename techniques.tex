\subsubsection{Exponential}
\label{subsubsec:probability-and-statistics:distributions:exponential}

\paragraph{PDF}
\label{para:probability-and-statistics:distributions:exponential:pdf}
\begin{equation}
    \label{eq:exponential-pdf}
    p(x; \lambda) =
    \begin{cases}
        \lambda e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{CDF}
\label{para:probability-and-statistics:distributions:exponential:cdf}
\begin{equation}
    \label{eq:exponential-cdf}
    P(x; \lambda) = P(X \leq x) =
    \begin{cases}
        1 - e^{-\lambda x}, & x \geq 0 \\
        0, & x < 0
    \end{cases}
\end{equation}

\paragraph{Mean}
\label{para:probability-and-statistics:distributions:exponential:mean}
\begin{equation}
    \label{eq:exponential-mean}
    \mathbb{E}[X] = \frac{1}{\lambda}
\end{equation}

\paragraph{Variance}
\label{para:probability-and-statistics:distributions:exponential:variance}
\begin{equation}
    \label{eq:exponential-variance}
    \operatorname{Var}(X) = \frac{1}{\lambda^2}
\end{equation}
